CREATE TABLE ASIGNATURA (ID_ASIGNATURA INT, NUM_ALUMNOS INT, LOCAL_CLASE VARCHAR(100), ID_ FORMADOR INT, CONSTRAINT  fk_asignatura_formador FOREIGN KEY (ID_FORMADOR) REFERENCES FORMADORES (ID_FORMADOR);


CREATE TABLE ESTUDIANTE (ID_ALUMNO INT, ID_ASIGNATURA VARCHAR(20), MATRICULA DATE, CONSTRAINT fk_estudiante_asignatura FOREIGN KEY (ID_ASIGNATURA) REFERENCES ASIGNATURA (ID_ASIGNATURA);
